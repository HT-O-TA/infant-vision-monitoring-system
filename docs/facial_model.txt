### **一、数据清洗与人脸对齐**
#### 1. **人脸检测与裁剪（RetinaFace）**
```python
import cv2
import os
from retinaface import RetinaFace

def detect_and_crop_faces(input_dir, output_dir, confidence_threshold=0.8):
    """
    使用RetinaFace检测人脸并裁剪保存
    Args:
        input_dir: 原始图像目录
        output_dir: 裁剪后人脸保存目录
        confidence_threshold: 置信度阈值
    """
    os.makedirs(output_dir, exist_ok=True)
    
    for img_name in os.listdir(input_dir):
        img_path = os.path.join(input_dir, img_name)
        img = cv2.imread(img_path)
        
        # 检测人脸
        faces = RetinaFace.detect_faces(img)
        if not faces:
            print(f"未检测到人脸，跳过: {img_name}")
            continue
        
        # 选择置信度最高的人脸
        best_face = max(faces, key=lambda x: x['confidence'])
        if best_face['confidence'] < confidence_threshold:
            print(f"置信度过低，跳过: {img_name}")
            continue
        
        # 提取人脸区域并保存
        x1, y1, w, h = best_face['facial_area']
        face_img = img[y1:y1+h, x1:x1+w]
        face_img = cv2.resize(face_img, (112, 112))  # 统一尺寸
        cv2.imwrite(os.path.join(output_dir, img_name), face_img)
```

**使用示例**：
```python
detect_and_crop_faces(
    input_dir="raw_images",
    output_dir="aligned_faces",
    confidence_threshold=0.8
)
```

### **二、数据标注流程**
#### 1. **半自动标注（DeepFace预标注）**
```python
from deepface import DeepFace
import os
import pandas as pd

def auto_annotate(input_dir, output_csv):
    """
    使用DeepFace预标注情绪，生成CSV文件
    Args:
        input_dir: 对齐后的人脸图像目录
        output_csv: 输出标注文件路径
    """
    data = []
    for img_name in os.listdir(input_dir):
        img_path = os.path.join(input_dir, img_name)
        try:
            # 使用DeepFace分析情绪
            result = DeepFace.analyze(img_path, actions=['emotion'], enforce_detection=False)
            label = result[0]['dominant_emotion']
            data.append({"image_path": img_path, "label": label})
        except Exception as e:
            print(f"标注失败 {img_name}: {str(e)}")
    
    # 保存为CSV
    df = pd.DataFrame(data)
    df.to_csv(output_csv, index=False)
```

**使用示例**：
```python
auto_annotate(
    input_dir="aligned_faces",
    output_csv="pre_annotate.csv"
)
```

#### 2. **YOLO格式转换**
```python
import os
import pandas as pd
from pathlib import Path

def csv_to_yolo(csv_path, output_dir, img_dir, class_mapping):
    """
    将CSV标注转换为YOLO格式
    Args:
        csv_path: 预标注CSV路径
        output_dir: 输出目录（含images和labels子目录）
        img_dir: 原始图像目录
        class_mapping: 类别ID映射字典（如{"happy":0, "sad":1}）
    """
    os.makedirs(os.path.join(output_dir, "images"), exist_ok=True)
    os.makedirs(os.path.join(output_dir, "labels"), exist_ok=True)
    
    df = pd.read_csv(csv_path)
    
    for _, row in df.iterrows():
        img_path = os.path.join(img_dir, row["image_path"])
        label = row["label"]
        class_id = class_mapping[label]
        
        # 生成YOLO格式标注
        h, w, _ = cv2.imread(img_path).shape
        x_center = 0.5  # 假设人脸居中
        y_center = 0.5
        width = 0.8     # 假设人脸占图像宽度的80%
        height = 0.3    # 假设人脸占图像高度的30%
        
        with open(os.path.join(output_dir, "labels", f"{Path(img_path).stem}.txt"), 'a') as f:
            f.write(f"{class_id} {x_center} {y_center} {width} {height}\n")
        
        # 复制图像到输出目录
        os.symlink(img_path, os.path.join(output_dir, "images", os.path.basename(img_path)))
```

**使用示例**：
```python
class_mapping = {"happy":0, "sad":1, "angry":2, "neutral":3, "surprised":4}
csv_to_yolo(
    csv_path="pre_annotate.csv",
    output_dir="dataset",
    img_dir="aligned_faces",
    class_mapping=class_mapping
)
```
